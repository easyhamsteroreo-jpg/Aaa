            <input type="text" id="toAddress" placeholder="Адрес доставки" value="ул. Независимости, 10, Киев">

            <button id="updateLocationBtn" class="secondary">📡 Отправить текущую позицию курьера</button>

<script>
    // --- Хранилище (localStorage) ---
    let parcels = [];     // массив посылок
    let courierLocation = { lat: 50.4501, lng: 30.5234, updated: null }; // Киев по умолчанию

    function loadData() {
        const stored = localStorage.getItem('easy_post_data');
        if (stored) {
            const data = JSON.parse(stored);
            parcels = data.parcels || [];
            courierLocation = data.courierLocation || { lat: 50.4501, lng: 30.5234, updated: null };
        } else {
            // демо-посылки
            parcels = [
                { id: "EP240512001UA", from: "Иван Петров", to: "Анна Сидорова", address: "ул. Независимости, 10, Киев", weight: "1.5", status: "created", courierNote: null, locationHistory: [] },
                { id: "EP240512002UA", from: "Олег Коваль", to: "Мария Лис", address: "пр. Свободы, 45, Львов", weight: "3.2", status: "office", courierNote: null, locationHistory: [] },
                { id: "EP240511455UA", from: "Сергей Бондар", to: "Тарас Шевченко", address: "ул. Грушевского, 12, Киев", weight: "0.8", status: "courier", courierNote: "Курьер: Андрей", locationHistory: [] }
            ];
            saveData();
        }
        renderAll();
    }

    function saveData() {
        localStorage.setItem('easy_post_data', JSON.stringify({ parcels, courierLocation }));
    }

    // Генерация трек-номера
    function generateTrackingNumber() {
        const letters = "ABCDEFGHJKLMNPQRSTUVWXYZ";
        const randomStr = Array.from({ length: 4 }, () => letters[Math.floor(Math.random() * letters.length)]).join('');
        const numPart = Math.floor(Math.random() * 1000000).toString().padStart(6, '0');
        return `EP${numPart}${randomStr}`;
    }

    // Создание посылки (отправитель)
    function createParcel() {
        const fromName = document.getElementById('fromName').value.trim();
        const toName = document.getElementById('toName').value.trim();
        const address = document.getElementById('toAddress').value.trim();
        const weight = document.getElementById('weight').value.trim();
        if (!fromName || !toName || !address) {
            alert("Заполните все поля");
            return;
        }
        const newParcel = {
            id: generateTrackingNumber(),
            from: fromName,
            to: toName,
            address: address,
            weight: weight || "0",
            status: "created",   // created → office → courier → delivered
            courierNote: null,
            locationHistory: []
        };
        parcels.unshift(newParcel);
        saveData();
        renderAll();
        // очистка формы
        document.getElementById('fromName').value = '';
        document.getElementById('toName').value = '';
        document.getElementById('toAddress').value = '';
        document.getElementById('weight').value = '';
        alert(`Посылка создана! Трек-номер: ${newParcel.id}`);
    }

    // Изменение статуса (работник почты или курьер)
    function updateStatus(parcelId, newStatus, courierName = null) {
        const parcel = parcels.find(p => p.id === parcelId);
        if (!parcel) return;
        parcel.status = newStatus;
        if (newStatus === "courier" && courierName) {
            parcel.courierNote = `Курьер: ${courierName}`;
        }
        if (newStatus === "delivered") {
            parcel.locationHistory.push({ action: "Доставлена", time: new Date().toLocaleString() });
        }
        saveData();
        renderAll();
    }

    // Курьер: обновление геолокации
    function updateCourierLocation() {
        if ("geolocation" in navigator) {
            navigator.geolocation.getCurrentPosition(
                (pos) => {
                    courierLocation = {
                        lat: pos.coords.latitude,
                        lng: pos.coords.longitude,
                        updated: new Date().toLocaleString()
                    };
                    saveData();
                    document.getElementById('lastLocation').innerHTML = `📍 Широта: ${courierLocation.lat.toFixed(5)}, Долгота: ${courierLocation.lng.toFixed(5)}<br>Время: ${courierLocation.updated}`;
                    alert("Геолокация курьера обновлена");
                    renderAll(); // обновим список посылок, чтобы показать, где курьер
                },
                (err) => {
                    alert("Не удалось получить геолокацию: " + err.message);
                    // симуляция
                    courierLocation = { lat: 50.4501 + (Math.random() - 0.5) * 0.02, lng: 30.5234 + (Math.random() - 0.5) * 0.02, updated: new Date().toLocaleString() };
                    saveData();
                    document.getElementById('lastLocation').innerHTML = `📍 Симуляция: ${courierLocation.lat.toFixed(5)}, ${courierLocation.lng.toFixed(5)}<br>Время: ${courierLocation.updated}`;
                    renderAll();
                }
            );
        } else {
            alert("Геолокация не поддерживается браузером");
        }
    }

    // Рендер всех ролей
    function renderAll() {
        renderSender();
        renderWorker();
        renderCourier();
    }

    // Отправитель: показывает его посылки
    function renderSender() {
        const container = document.getElementById('senderParcelsList');
        const myParcels = parcels.filter(p => p.from === document.getElementById('fromName').value.trim() || true); // в демо-режиме показываем все, но по идее по отправителю
        // для удобства показываем все посылки отправителя (по полю from, если оно совпадает с текущим вводом, но лучше показывать все созданные текущим пользователем — упростим: все)
        // реально надо связывать с авторизацией — сделаем проще: отображаем все посылки со статусом
        if (parcels.length === 0) {
            container.innerHTML = '<div class="parcel-item">Нет посылок. Создайте первую.</div>';
            return;
        }
        let html = '';
        parcels.forEach(p => {
            let statusText = "";
            if (p.status === "created") statusText = "Создана (ожидает приёма)";
            else if (p.status === "office") statusText = "Принята в отделении";
            else if (p.status === "courier") statusText = "У курьера";
            else statusText = "Доставлена";
            html += `
                <div class="parcel-item">
                    <div class="flex-between">
                        <span class="parcel-id">${p.id}</span>
                        <span class="status status-${p.status}">${statusText}</span>
                    </div>
                    <div>📤 От: ${p.from}</div>
                    <div>📥 Кому: ${p.to}</div>
                    <div>📍 Адрес: ${p.address}</div>
                    <div>⚖️ Вес: ${p.weight} кг</div>
                </div>
            `;
        });
        container.innerHTML = html;
    }

    // Работник почты: может менять статус created → office → courier
    function renderWorker() {
        const container = document.getElementById('workerParcelsList');
        if (parcels.length === 0) {
            container.innerHTML = '<div class="parcel-item">Нет посылок в системе.</div>';
            return;
        }
        let html = '';
        parcels.forEach(p => {
            let statusText = "";
            if (p.status === "created") statusText = "Создана";
            else if (p.status === "office") statusText = "На складе";
            else if (p.status === "courier") statusText = "У курьера";
            else statusText = "Доставлена";
            html += `
                <div class="parcel-item">
                    <div class="flex-between">
                        <span class="parcel-id">${p.id}</span>
                        <span class="status status-${p.status}">${statusText}</span>
                    </div>
                    <div>📤 От: ${p.from}</div>
                    <div>📥 Кому: ${p.to}</div>
                    <div>📍 Адрес: ${p.address}</div>
                    <div>⚖️ ${p.weight} кг</div>
                    <div style="margin-top: 8px;">
            `;
            // Кнопки действий в зависимости от статуса
            if (p.status === "created") {
                html += `<button class="secondary" onclick="updateStatus('${p.id}', 'office')">📥 Принять на склад</button>`;
            } else if (p.status === "office") {
                html += `<button class="secondary" onclick="updateStatus('${p.id}', 'courier', 'Курьер №1')">🚚 Передать курьеру</button>`;
            } else if (p.status === "courier") {
                html += `<button class="secondary" disabled style="opacity:0.5;">⏳ В курьерской доставке</button>`;
            } else {
                html += `<button disabled style="background:#e2e3e5;">✅ Доставлено</button>`;
            }
            html += `</div></div>`;
        });
        container.innerHTML = html;
    }

    // Курьер: видит посылки со статусом "courier" и может отметить доставку
    function renderCourier() {
        const container = document.getElementById('courierParcelsList');
        const courierParcels = parcels.filter(p => p.status === "courier");
        if (courierParcels.length === 0) {
            container.innerHTML = '<div class="parcel-item">Нет назначенных посылок.</div>';
            return;
        }
        let html = '';
        courierParcels.forEach(p => {
            html += `
                <div class="parcel-item">
                    <div class="flex-between">
                        <span class="parcel-id">${p.id}</span>
                        <span class="status status-courier">У курьера</span>
                    </div>
                    <div>📥 Получатель: ${p.to}</div>
                    <div>📍 Адрес: ${p.address}</div>
                    <div>⚖️ ${p.weight} кг</div>
                    <button class="secondary" onclick="updateStatus('${p.id}', 'delivered')">✅ Доставлено</button>
                </div>
            `;
        });
        container.innerHTML = html;

        // Показываем последнюю геолокацию курьера
        if (courierLocation.updated) {
            document.getElementById('lastLocation').innerHTML = `📍 Последняя позиция курьера:<br>Широта: ${courierLocation.lat.toFixed(5)}, Долгота: ${courierLocation.lng.toFixed(5)}<br>Время: ${courierLocation.updated}`;
        } else {
            document.getElementById('lastLocation').innerHTML = `📍 Позиция ещё не отправлена. Нажмите кнопку выше.`;
        }
    }

    // --- Переключение ролей ---
    function switchRole(role) {
        document.querySelectorAll('.panel').forEach(panel => panel.classList.remove('active-panel'));
        if (role === 'sender') document.getElementById('senderPanel').classList.add('active-panel');
        if (role === 'worker') document.getElementById('workerPanel').classList.add('active-panel');
        if (role === 'courier') document.getElementById('courierPanel').classList.add('active-panel');
        
        document.querySelectorAll('.role-btn').forEach(btn => btn.classList.remove('active'));
        document.querySelector(`.role-btn[data-role="${role}"]`).classList.add('active');
        renderAll(); // обновить данные при смене
    }

    // Инициализация событий
    document.getElementById('createParcelBtn').addEventListener('click', createParcel);
    document.getElementById('updateLocationBtn').addEventListener('click', updateCourierLocation);
    
    document.querySelectorAll('.role-btn').forEach(btn => {
        btn.addEventListener('click', (e) => {
            switchRole(e.target.getAttribute('data-role'));
        });
    });

    // Загружаем данные и рендерим
    loadData();
    window.updateStatus = updateStatus; // для вызова из onclick
</script>
